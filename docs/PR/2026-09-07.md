  postgres redis docker-socket-proxy api web nlp_intent office slack_baneo_worker botellas_recalculo_worker cromo_worker
docker cp reconciliar_baneos_prod.py lasfocas-web:/tmp/
docker exec -e PYTHONPATH=/app -w /app lasfocas-web python3 /tmp/reconciliar_baneos_prod.py --apply
docker exec lasfocas-web rm -f /tmp/reconciliar_baneos_prod.py /tmp/prod_baneos_snapshot.json
